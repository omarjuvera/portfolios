sudo reboot

=======================================
Step 2: Install Apache Web Server
=======================================
sudo apt install apache2 -y
sudo systemctl enable apache2
sudo systemctl status apache2 --no-pager

	sudo ufw app list
sudo ufw allow 'Apache Secure'
sudo ufw app info 'Apache Secure'
	OR

		sudo ufw app list
	sudo ufw allow 'Apache Full'
		allows both incoming and outgoing traffic on the required ports (80 for HTTP and 443 for HTTPS).
	sudo ufw app info 'Apache Full'
		make sure ports [80,443/tcp] are listed

sudo ufw enable

sudo apt update -y && sudo apt upgrade -y
	Run this as many times as needed until you see this:
	0 upgraded, 0 newly installed, 0 to remove and 0 not upgraded.

sudo systemctl restart apache2 && sudo systemctl status apache2 --no-pager

=======================================
Step 5: Configure Apache to work with PHP
=======================================
sudo nano /etc/apache2/mods-enabled/dir.conf
	Ensure index.php is listed first (right after DirectoryIndex)
	DirectoryIndex index.php index.html index.cgi index.pl index.xhtml index.htm
sudo systemctl restart apache2 && sudo systemctl status apache2 --no-pager

=======================================
Step 6: Enable HTTPS (via Apache configuration)
=======================================
sudo a2enmod ssl
sudo a2ensite default-ssl
sudo apt install certbot python3-certbot-apache -y
sudo certbot --apache

	See /usr/share/doc/apache2/README.Debian.gz on how to configure SSL and create self-signed certificates.

	Simplified, for beguiners:
sudo apt install certbot python3-certbot-apache -y
sudo certbot --apache
sudo systemctl reload apache2
sudo systemctl status apache2 --no-pager

===
Step 6.3: Redirect HTTP to HTTPS:
===
sudo nano /etc/apache2/sites-available/000-default.conf

Add a redirect rule at the end of the file before </VirtualHost>:

<VirtualHost *:80>
    Redirect permanent / https://your-server-ip/
</VirtualHost>

	Replace your-server-ip with your actual IP or domain.

sudo systemctl restart apache2
sudo systemctl status apache2 --no-pager
